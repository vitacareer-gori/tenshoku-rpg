      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
    }

    function toggleAudio() {
      audioEnabled = !audioEnabled;
      document.getElementById('audioState').textContent = audioEnabled ? 'ON' : 'OFF';
      document.getElementById('audioIcon').textContent = audioEnabled ? '🔊' : '🔈';
    }

    function playSfx(type) {
      if (!audioEnabled) return;
      try {
        initAudio();
        const now = audioCtx.currentTime;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        if (type === 'enter') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(320, now);
          osc.frequency.exponentialRampToValueAtTime(740, now + 0.15);
          gain.gain.setValueAtTime(0.2, now);
          gain.gain.linearRampToValueAtTime(0.01, now + 0.25);
          osc.start(now);
          osc.stop(now + 0.25);
        } else if (type === 'check') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(523.25, now);
          osc.frequency.setValueAtTime(659.25, now + 0.08);
          osc.frequency.setValueAtTime(783.99, now + 0.16);
          osc.frequency.setValueAtTime(1046.50, now + 0.24);
          gain.gain.setValueAtTime(0.18, now);
          gain.gain.linearRampToValueAtTime(0.01, now + 0.45);
          osc.start(now);
          osc.stop(now + 0.45);
        }
      } catch (e) {}
    }

    // --- 進捗管理 (localStorage) ---
    const STORAGE_KEY = 'rpg_career_cleared_stages';
    const totalStages = 7;

    function getClearedStages() {
      const stored = localStorage.getItem(STORAGE_KEY);
      return stored ? JSON.parse(stored) : [];
    }

    function saveClearedStages(stages) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(stages));
      updateUI();
    }

    function toggleQuest(stageId) {
      let stages = getClearedStages();
      const sId = String(stageId);
      if (stages.includes(sId)) {
        stages = stages.filter(id => id !== sId);
      } else {
        stages.push(sId);
        playSfx('check');
      }
      saveClearedStages(stages);
    }

    function resetProgress() {
      if (confirm('すべての進行度をリセットしますか？')) {
        localStorage.removeItem(STORAGE_KEY);
        updateUI();
      }
    }

    function updateUI() {
      const stages = getClearedStages();
      const count = stages.length;
      const pct = Math.round((count / totalStages) * 100);

      document.getElementById('progressBar').style.width = pct + '%';
      document.getElementById('progressPercentage').textContent = pct + '% 攻略完了';
      document.getElementById('progressCount').textContent = `達成クエスト: ${count} / ${totalStages}`;

      let rank = '見習い冒険者';
      if (pct === 100) rank = '伝説の英雄 (内定・新天地獲得)';
      else if (pct >= 70) rank = '百戦錬磨の勇者';
      else if (pct >= 40) rank = '中堅の旅人';
      else if (pct > 0) rank = '志を持つ探求者';
      document.getElementById('rankBadge').textContent = `RANK: ${rank}`;

      document.querySelectorAll('.quest-card').forEach(card => {
        const stage = card.getAttribute('data-stage');
        const btn = card.querySelector('.btn-check');
        if (stages.includes(stage)) {
          card.classList.add('completed');
          btn.innerHTML = '<span class="check-icon">✅</span> 踏破完了';
        } else {
          card.classList.remove('completed');
          btn.innerHTML = '<span class="check-icon">⬜</span> 未踏破';
        }
      });
    }

    // --- 風・花びら・光の粒子アニメーション (Canvas) ---
    const canvas = document.getElementById('windCanvas');
    const ctx = canvas.getContext('2d');

    let width, height;
    function resizeCanvas() {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    const particleCount = 45;
    const particles = [];

    class Particle {
      constructor() {
        this.reset(true);
      }
      reset(init = false) {
        this.x = init ? Math.random() * width : -20;
        this.y = Math.random() * height;
        this.size = Math.random() * 3.5 + 1.2;
        this.vx = Math.random() * 2.5 + 1.2;
        this.vy = (Math.random() - 0.3) * 0.8;
        this.alpha = Math.random() * 0.6 + 0.2;
        this.type = Math.random() > 0.4 ? 'petal' : 'spore'; // 花びら or 光
        this.rot = Math.random() * 360;
        this.vRot = (Math.random() - 0.5) * 2;
      }
      update() {
        this.x += this.vx;
        this.y += this.vy;
        this.rot += this.vRot;
        if (this.x > width + 30 || this.y < -30 || this.y > height + 30) {
          this.reset();
        }
      }
      draw() {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate((this.rot * Math.PI) / 180);
        ctx.globalAlpha = this.alpha;

        if (this.type === 'petal') {
          // ボルドー・ローズ系の花びら
          ctx.fillStyle = '#b82d49';
          ctx.beginPath();
          ctx.ellipse(0, 0, this.size * 2, this.size, 0, 0, Math.PI * 2);
          ctx.fill();
        } else {
          // 黄金・ルビーの光彩
          ctx.fillStyle = '#f5d08c';
          ctx.shadowBlur = 8;
          ctx.shadowColor = '#e63255';
          ctx.beginPath();
          ctx.arc(0, 0, this.size, 0, Math.PI * 2);
          ctx.fill();
        }
        ctx.restore();
      }
    }

    const windLines = Array.from({ length: 6 }, () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      len: Math.random() * 120 + 80,
      speed: Math.random() * 4 + 3,
      alpha: Math.random() * 0.15 + 0.05
    }));

    for (let i = 0; i < particleCount; i++) {
      particles.push(new Particle());
    }

    function animate() {
      ctx.clearRect(0, 0, width, height);

      ctx.strokeStyle = 'rgba(255, 230, 235, 0.15)';
      ctx.lineWidth = 1;
      windLines.forEach(w => {
        ctx.globalAlpha = w.alpha;
        ctx.beginPath();
        ctx.moveTo(w.x, w.y);
        ctx.lineTo(w.x + w.len, w.y - w.len * 0.1);
        ctx.stroke();
        w.x += w.speed;
        if (w.x > width + w.len) {
          w.x = -w.len;
          w.y = Math.random() * height;
        }
      });

      particles.forEach(p => {
        p.update();
        p.draw();
      });

      requestAnimationFrame(animate);
    }
    animate();

    document.addEventListener('DOMContentLoaded', () => {
      updateUI();
    });
  </script>
